"""Behaviour of template_name: which posts it points at, and why. Replace these expectations with yours."""
from pathlib import Path
from commsfail.annotators import registry
from commsfail.sources import sharednet

GOAL_RUN = Path(__file__).resolve().parents[1] / "fixtures" / "goal_run"

def test_template_name_on_the_synthetic_room(synthetic_trace):
    out = registry()["template_name"]().annotate(synthetic_trace)
    assert [f["seq"] for f in out["findings"]] == [2, 11]       # two questions that nobody answers or cites

def test_template_name_on_the_goal_run_sample():
    out = registry()["template_name"]().annotate(sharednet.load(str(GOAL_RUN)))
    assert [f["seq"] for f in out["findings"]] == [6]           # codex-2 asks codex-1; codex-1 never answers
